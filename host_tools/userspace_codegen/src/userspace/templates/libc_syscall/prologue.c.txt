static unsigned long aethercore_append_char(char *dst, unsigned long pos, unsigned long cap, char ch) {
    if (cap != 0 && pos + 1 < cap) {
        dst[pos] = ch;
    }
    return pos + 1;
}

static unsigned long aethercore_append_str(char *dst, unsigned long pos, unsigned long cap, const char *src) {
    unsigned long i = 0;
    if (!src) src = "(null)";
    while (src[i] != 0) {
        pos = aethercore_append_char(dst, pos, cap, src[i]);
        ++i;
    }
    return pos;
}

static unsigned long aethercore_append_uint(char *dst, unsigned long pos, unsigned long cap, unsigned long value, unsigned base, int uppercase) {
    char tmp[32];
    unsigned long i = 0;
    const char *digits = uppercase ? "0123456789ABCDEF" : "0123456789abcdef";
    if (base < 2) base = 10;
    do {
        tmp[i++] = digits[value % base];
        value /= base;
    } while (value != 0 && i < sizeof(tmp));
    while (i != 0) {
        pos = aethercore_append_char(dst, pos, cap, tmp[--i]);
    }
    return pos;
}

int vsnprintf(char *dst, unsigned long size, const char *fmt, va_list ap) {
    unsigned long pos = 0;
    unsigned long i = 0;
    if (!fmt) {
        if (size != 0 && dst) dst[0] = 0;
        return 0;
    }
    while (fmt[i] != 0) {
        if (fmt[i] != '%') {
            pos = aethercore_append_char(dst, pos, size, fmt[i++]);
            continue;
        }
        ++i;
        switch (fmt[i]) {
        case '%':
            pos = aethercore_append_char(dst, pos, size, '%');
            break;
        case 's':
            pos = aethercore_append_str(dst, pos, size, va_arg(ap, const char *));
            break;
        case 'c':
            pos = aethercore_append_char(dst, pos, size, (char)va_arg(ap, int));
            break;
        case 'd':
        case 'i': {
            long value = va_arg(ap, int);
            if (value < 0) {
                pos = aethercore_append_char(dst, pos, size, '-');
                pos = aethercore_append_uint(dst, pos, size, (unsigned long)(-value), 10, 0);
            } else {
                pos = aethercore_append_uint(dst, pos, size, (unsigned long)value, 10, 0);
            }
            break;
        }
        case 'u':
            pos = aethercore_append_uint(dst, pos, size, va_arg(ap, unsigned int), 10, 0);
            break;
        case 'x':
            pos = aethercore_append_uint(dst, pos, size, va_arg(ap, unsigned int), 16, 0);
            break;
        case 'X':
            pos = aethercore_append_uint(dst, pos, size, va_arg(ap, unsigned int), 16, 1);
            break;
        case 'p': {
            unsigned long value = (unsigned long)va_arg(ap, void *);
            pos = aethercore_append_str(dst, pos, size, "0x");
            pos = aethercore_append_uint(dst, pos, size, value, 16, 0);
            break;
        }
        default:
            pos = aethercore_append_char(dst, pos, size, '%');
            if (fmt[i] != 0) {
                pos = aethercore_append_char(dst, pos, size, fmt[i]);
            }
            break;
        }
        if (fmt[i] != 0) ++i;
    }
    if (size != 0 && dst) {
        dst[pos < size ? pos : size - 1] = 0;
    }
    return (int)pos;
}

int snprintf(char *dst, unsigned long size, const char *fmt, ...) {
    va_list ap;
    int ret;
    va_start(ap, fmt);
    ret = vsnprintf(dst, size, fmt, ap);
    va_end(ap);
    return ret;
}

int printf(const char *fmt, ...) {
    char buffer[1024];
    va_list ap;
    int len;
    va_start(ap, fmt);
    len = vsnprintf(buffer, sizeof(buffer), fmt, ap);
    va_end(ap);
    if (len < 0) {
        return -1;
    }
    if (write(1, buffer, (unsigned long)len) < 0) {
        return -1;
    }
    return len;
}

int puts(const char *s) {
    long ret;
    if (!s) {
        *__errno_location() = 22;
        return -1;
    }
    ret = write(1, s, strlen(s));
    if (ret < 0) {
        return -1;
    }
    if (write(1, "\n", 1) < 0) {
        return -1;
    }
    return (int)(ret + 1);
}

int aethercore_write_cstr_fd(const char *s, int fd) {
    long ret;
    if (!s) {
        *__errno_location() = 22;
        return -1;
    }
    ret = write(fd, s, strlen(s));
    return ret < 0 ? -1 : (int)ret;
}

void perror(const char *prefix) {
    const char *message = strerror(*__errno_location());
    if (prefix && prefix[0] != 0) {
        (void)aethercore_write_cstr_fd(prefix, 2);
        (void)aethercore_write_cstr_fd(": ", 2);
    }
    (void)aethercore_write_cstr_fd(message, 2);
    (void)write(2, "\n", 1);
}
