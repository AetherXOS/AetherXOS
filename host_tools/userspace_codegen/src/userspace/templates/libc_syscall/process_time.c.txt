unsigned int sleep(unsigned int seconds) {
    struct aethercore_timespec req;
    struct aethercore_timespec rem;
    req.tv_sec = (long)seconds;
    req.tv_nsec = 0;
    if (nanosleep(&req, &rem) == 0) {
        return 0;
    }
    return rem.tv_sec > 0 ? (unsigned int)rem.tv_sec : 0u;
}

int usleep(unsigned long usec) {
    struct aethercore_timespec req;
    req.tv_sec = (long)(usec / 1000000ul);
    req.tv_nsec = (long)((usec % 1000000ul) * 1000ul);
    return nanosleep(&req, (struct aethercore_timespec *)0);
}
